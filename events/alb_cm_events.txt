#######################################################################################################################
###These events are used as countermeasures should the player or AI manage to convert to the ALB faiths

## alb_cm.0001: Main event, this explains to the player why the faith exists and why they can't play as it.
#   -option a: this option sends the player to a list of Islamic faiths to choose from.
#   -option b: this option converts the player to their capital's faith
#   -option c: This option kills the character.

## alb_cm.0002: Main list, this event creates a list of options that lead to a secondary list based on the succession doctrine.
#   -option a: Sunni option, leads to an event that lists all sunni faiths that the player can convert to.
#   -option b: Shia option, leads to an event that lists all shia faiths that the player can convert to.
#   -option c: Batriya option, leads to an event that lists all batriya faiths that the player can convert to.
#   -option d: Muhakkima option, leads to an event that lists all muhakkima faiths that the player can convert to.
#   -option e: Zandaqa option, leads to an event that lists all zandaqa faiths that the player can convert to.
#   -option f: Ghulat option, leads to an event that lists all ghulat faiths that the player can convert to.
#   -option g: Sends the player to the previous event.

## alb_cm.0003: Sunni list, this lists all sunni faiths that the player can convert to. 
#   -option a: Ahl al-Sunnah (ashari)
#   -option b: Muwalladi
#   -option c: Maturidi
#   -option d: Masmudi (Almohad), doesn't appear until 1130
#   -option f: Sends the player to the previous event.

## alb_cm.0004: Shia list, this lists all shia faiths that the player can convert to. 
#   -option a: Isma'ili
#   -option b: Nizari
#   -option c: Zayidi
#   -option d: Itha 'Ashri (imami)
#   -option e: Hafizi
#   -option f: Sends the player to the previous event.

## alb_cm.0005: Batriya list, this lists all batriya faiths that the player can convert to. 
#   -option a: Salihism
#   -option b: Sends the player to the previous event.

## alb_cm.0006: Muhakkima list, this lists all muhakkima faiths that the player can convert to. 
#   -option a: Ibadi
#   -option b: Wahbiya
#   -option c: Nukkari
#   -option d: Azariqa
#   -option e: Najdat
#   -option f: Sufri
#   -option g: Sends the player to the previous event.

## alb_cm.0007: Zandaqa list, this lists all zandaqa faiths that the player can convert to.
#   -option a: Mutazila 
#   -option b: Qur'anist
#   -option c: Nazzamism (REMOVED)
#   -option d: Barghawata
#   -option e: Sends the player to the previous event.

## alb_cm.0008: Ghulat list, this lists all ghulat faiths that the player can convert to. 
#   -option a: Qarmatian
#   -option b: Ikhtilafism (ghulat)
#   -option c: Alawite
#   -option d: Ishaqiya
#   -option e: Alevi
#   -option f: Druze
#   -option g: Sends the player to the previous event.
#######################################################################################################################

namespace = alb_cm

#Main Event
alb_cm.0001 = {
	title = alb_cm.0001.t
	desc = {
		desc = alb_cm.0001.desc.intro
		first_valid = {
			triggered_desc = {
				trigger = { root.faith = faith:ns_islam }
				desc = alb_cm.0001.desc.main
			}
			triggered_desc = {
				trigger = { root.faith = faith:ns_islam_shia }
				desc = alb_cm.0001.desc.shia
			}
		}
	}
	theme = faith
	background = study
	
	left_portrait = {
		character = root
		animation = personality_zealous
	}

	#Convert to different Islamic faith from list
	option = {
		name = alb_cm.0001.a
		custom_tooltip = alb_cm.0001.tt.showlist

		trigger = {
			is_ai = no
		}

		trigger_event = alb_cm.0002
	}

	#Convert to faith of capital
	option = {
		name = alb_cm.0001.b

		#Option shoudn't appear if capital is an ALB religion
		trigger = {
			is_ai = no
			is_landed = yes
			root.capital_county.faith = { NOT = { religion_tag = alb_religion } }
		}

		set_character_faith = root.capital_county.faith
	}

	#Kill off the character
	option = {
		name = alb_cm.0001.c

		trigger = {
			is_ai = no
		}

		death = {
			death_reason = death_vanished
		}
	}
}
