## Main Events

# Decision Taken
form_maltese.0001 = {
	title = form_maltese.0001.t
	desc = form_maltese.0001.desc
	theme = realm

# Maltese successfully formed
form_maltese.0002 = {
	title = form_maltese.0002.t
	theme = realm
	override_background = { reference = throne_room }
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:maltese_former = root
				}
				desc = form_maltese.0002.desc_former
			}
			triggered_desc = {
				trigger = {
					AND = {
						religion = religion:islam_religion
						NOT = { has_trait = cynical }
					}
				}
				desc = form_maltese.0002.desc_islam
			}
			triggered_desc = {
				trigger = {
					NOR = {
						scope:maltese_former = root
						AND = {
							religion = religion:islam_religion
							NOT = { has_trait = cynical }
						}
					}
				}
				desc = form_maltese.0002.desc_norm
			}
		}
	}
	
	
    left_portrait = {
    	character = root
    	triggered_animation = {
			trigger = { scope:maltese_former = root }
			animation = personality_bold
		}
		triggered_animation = {
			trigger = {
				AND = { 
					religion = religion:islam_religion
					NOT = { has_trait = cynical }
				}
			}
			animation = disapproval
		}
		triggered_animation = {
			trigger = { always = yes }
			animation = personality_rational
		}
    }
	
	immediate = {
		show_as_tooltip = { me_maltese_culture_forms_effects = yes } ## Actually happened in form_maltese.0010
	}
	
	option = {
		name = form_maltese.0002.a
		custom_tooltip = form_maltese.0002.tt
		
	}
	
	option = {
		trigger = { capital_county = title:c_malta }
		name = form_maltese.0002.b
		custom_tooltip = form_maltese.0002.tt
		
		convert_family_culture_and_notify_vassals_effect = {
			CONVERTER = root
			OLD_CULTURE = root.culture
			NEW_CULTURE = capital_province.culture
		}
		
		ai_chance = {
			base = 1000
		}
	}
}

## Check events

# Check if Malta is still Christian
form_maltese.0010 = {
	hidden = yes

	immediate = {
		if = {
			limit = {
				title:c_malta.faith.religion = religion:christianity_religion
				title:c_malta.culture = {
					AND = {
						has_cultural_pillar = heritage_arabic
						has_cultural_pillar = language_arabic
					}
				}
			}
			me_maltese_culture_forms_effects = yes
			trigger_event = {
				id = form_maltese.0011
				days = 36500 #100 years
			}
			every_player = {
				limit = {
					completely_controls = title:c_malta
				}
				trigger_event = form_maltese.0002
			}
			title:c_malta.holder = {
				if = {
					limit = { is_ai = yes }
					trigger_event = form_maltese.0002
				}
			}
		}
	}
}

# Check if Maltese needs the Maltese language
form_maltese.0011 = {
	hidden = yes

	immediate = {
		if ={
			limit = {
				culture:maltese = { NOT = { has_cultural_pillar = language_maltese } }
			}
			me_maltese_language_forms_effects = yes
			every_player = {
				limit = {
					OR = {
						capital_county.culture = culture:maltese
						culture = culture:maltese
					}
				}
				trigger_event = form_maltese.0003
			}
		}
	}
}
