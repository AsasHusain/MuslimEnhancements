scripted_trigger find_hafidh_basic_requirements_trigger = {
	is_adult = yes
	is_imprisoned = no
	is_alive = yes # Because apparently there can be dead characters in the pool (remove when fixed)
	is_claimant = no # We don't want recruiting tutors to be a cheap way of getting claimants
	learning >= low_skill_rating
	has_trait = hafidh
	AND = {
		faith.religion = religion:islam_religion
		NOT = { faith = { has_doctrine = muhammad_succession_berberist_doctrine } }
		NOT = { faith = faith:druze }
	}
}

scripted_trigger find_hafidh_excellent_skill_tutor_trigger = {
	find_hafidh_basic_requirements_trigger = yes
	learning >= high_skill_rating # 15
}

scripted_trigger find_hafidh_high_skill_tutor_trigger = {
	find_hafidh_basic_requirements_trigger = yes
	#Empire/kingdom
	trigger_if = {
		limit = {
			root = { highest_held_title_tier >= tier_kingdom }
		}
		learning <= very_high_skill_rating # 18
		learning >= decent_skill_rating # 12
	}
	#Duchy/county
	trigger_else = {
		learning <= high_skill_rating  # 15
		learning >= medium_skill_rating  # 10
	}
}

scripted_trigger find_hafidh_low_skill_tutor_trigger = {
	find_hafidh_basic_requirements_trigger = yes
	#Empire/kingdom
	trigger_if = {
		limit = {
			root = { highest_held_title_tier >= tier_kingdom }
		}
		learning <= decent_skill_rating # 12
		learning >= mediocre_skill_rating # 8
	}
	#Duchy/county
	trigger_else = {
		learning <= medium_skill_rating # 10
		learning >= low_skill_rating # 5
	}
}

scripted_trigger find_hafidh_low_skill_tutor_light_trigger = {
	find_hafidh_basic_requirements_trigger = yes
	learning >= medium_skill_rating # 10
}

							find_hafidh_excellent_skill_tutor_trigger = yes

							knows_language = language_arabic
						}
						alternative_limit = {
							find_hafidh_excellent_skill_tutor_trigger = yes
							knows_language = language_arabic
						}

							base = 1
							modifier = {
								faith = root.faith
								factor = 10
							}
							modifier = {
								knows_language = language_arabic
								factor = 10
							}
							modifier = {
								has_trait = intellect_good
								factor = 10
							}
							modifier = {
								has_trait = zealous
								factor = 10
							}
							modifier = {
								has_trait = shrewd
								factor = 10
							}
							modifier = {
								has_trait = theologian
								factor = 10
							}
							modifier = {
								has_trait = scholar
								factor = 10
							}
						}

						# Knows the Arabic language
						if = {
							limit = {
								NOT = { knows_language = language_arabic }
							}
							learn_language_of_culture = culture:bedouin
						}
						if = {
							limit = {
								NOT = {
									has_trait_xp = {
										trait = hafidh
										value >= 30
									}
								}
							}
							add_trait_xp = {
				                trait = hafidh
				                value = {
				                    integer_range = {
				                        min = 30
				                        max = 100
				                    }
				                }

					}

							location = root.capital_province

							save_scope_as = excellent_skill_option
						}
					}

					# Ensure some ability of understanding the Qur'an
					scope:excellent_skill_option ?= {
						if = {
							limit = {
								NOT = { has_trait = hafidh }
							}
							add_trait = hafidh
							add_trait_xp = {
								trait = hafidh
								value = {
									65
									100
								}
							}
						}
						else_if = {
							limit = {
								has_trait_xp = {
									trait = hafidh
									value <= 50
								}
							}
							add_trait_xp = {
								trait = hafidh
								value = {
									30
									60
								}
							}
						}
			}
				}

				#HIGH-SKILL CHARACTER
				random_pool_character = {
					province = capital_province
					limit = {
						find_hafidh_high_skill_tutor_trigger = yes
						has_trait = hafidh
						NOT = { scope:excellent_skill_option ?= this }
					}
					alternative_limit = {
						find_hafidh_high_skill_tutor_trigger = yes
						NOT = { scope:excellent_skill_option ?= this }
					}
					weight = {
						base = 1
						modifier = {
							faith = root.faith
							factor = 10
						}
						modifier = {
							knows_language = language_arabic
							factor = 10
						}
						modifier = {
							has_trait = intellect_good
							factor = 10
						}
						modifier = {
							has_trait = zealous
							factor = 10
						}
						modifier = {
							has_trait = shrewd
							factor = 10
						}
						modifier = {
							has_trait = theologian
							factor = 10
						}
						modifier = {
							has_trait = scholar
							factor = 10
						}
					}
					save_scope_as = high_skill_option
				}

				if = {
					limit = { NOT = { exists = scope:high_skill_option } }
					create_character = {
						location = root.capital_province
						template = hafidh_high_character_template

						limit = {
							NOT = { has_trait = hafidh }
						}

							trait = hafidh

								10
								40
							}

					}
				}

				random_pool_character = {
					province = root.capital_province
					limit = { 
						find_hafidh_low_skill_tutor_trigger = yes
						has_trait = hafidh
						NOT = { scope:excellent_skill_option ?= this }

					}
					alternative_limit = {
						find_hafidh_low_skill_tutor_trigger = yes
						NOT = { scope:excellent_skill_option ?= this }

							faith = root.faith

							trait = hafidh
							value = {
								10
								30
							}
						}
					}
				}
				else_if = {
					limit = {
						has_trait_xp = {
							trait = hafidh
							value <= 5
						}
					}
					add_trait_xp = {
						trait = hafidh
						value = {
							10
							45
						}
					}
				}
			}

			exists = scope:excellent_skill_option

			}

				recipient = scope:excellent_skill_option

			}
		}
		else_if = {
			limit = {
				exists = court_position:court_tutor_court_position
			}
			court_position:court_tutor_court_position = {
				add_opinion = {
					target = root
					modifier = disappointed_opinion
					opinion = -5
				}
			}
			replace_court_position = {
				recipient = scope:excellent_skill_option
				holder = court_position:court_tutor_court_position
				court_position = court_tutor_court_position
			}
		}

			factor = 500

				short_term_gold < medium_gold_value

			}
		}
	}

	option = {

			is_landed = yes #extra check to make sure they haven't lost their status
			exists = scope:high_skill_option #Doesn't exists for AI
		}
		name = me_religious_decisions.0003.a

		custom_tooltip = me_religious_decisions.0003.a.aptitude
		pay_short_term_gold = {
			target = scope:high_skill_option
			gold = medium_gold_value
		}
		custom_tooltip = me_religious_decisions.0003.a.tt

		if = {
			limit = {
				can_employ_court_position_type = court_tutor_court_position
				has_royal_court = yes
			}
			appoint_court_position = {
				recipient = scope:high_skill_option
				court_position = court_tutor_court_position
			}
		}
		else_if = {
			limit = {
				exists = court_position:court_tutor_court_position
				has_royal_court = yes
			}
			court_position:court_tutor_court_position = {
				add_opinion = {
					target = root

			}

				holder = court_position:court_tutor_court_position

		}

	#Low
	option = {

			is_landed = yes #extra check to make sure they haven't lost their status
		}
		name = me_religious_decisions.0003.b

		custom_tooltip = me_religious_decisions.0003.b.aptitude
		if = {

				OR = {
					is_ai = no

				}
			}

				target = scope:low_skill_option
				gold = minor_gold_value

me_religious_decisions.0007 = {
	type = character_event
	title = me_religious_decisions.0007.t
	desc = me_religious_decisions.0007.desc
	theme = education
	left_portrait = scope:good_linguist
	lower_left_portrait = scope:cheap_linguist
	right_portrait = scope:medium_linguist
	lower_right_portrait = scope:scholarly_linguist
	override_background = { reference = throne_room }

	immediate = {
		### Save cultures and languages
		# Find 1st culture that speaks Arabic natively
		random_culture_global = {
			limit = {
				exists = culture_head
				has_cultural_pillar = language_arabic
			}
			save_scope_as = ling_cul_1
		}
		# Save liege as example to fetch faiths if relevant
		if = { 
			limit = {
				exists = scope:language_scheme_type
				liege = { has_court_language_of_culture = root.liege.culture }
			}
			liege = { save_scope_as = ling_cul_1_ruler }
		}
		# Otherwise find random cul_1 ruler
		else = {
			linguist_random_realm_effect = { CULTURE = ling_cul_1 }
		}
		# Find 2nd culture that speaks target language natively
		if = {
			limit = { 
				any_culture_global = {
					exists = culture_head
					has_cultural_pillar = language_arabic
					NOT = { THIS = scope:ling_cul_1 }
				}
			}
			random_culture_global = { 
				limit = {
					exists = culture_head
					has_cultural_pillar = language_arabic
					NOT = { THIS = scope:ling_cul_1 }
				}
				save_scope_as = ling_cul_2
			}
		}
		# If none, duplicate 1st
		else = {
			scope:ling_cul_1 = { save_scope_as = ling_cul_2 }
		}
		# Save random ruler of cul 2 to find faith
		linguist_random_realm_effect = { CULTURE = ling_cul_2 }
		# Save own culture
		culture = { save_scope_as = ling_cul_3 }
		# Bonus languages good linguists may know in addition to root and target
		linguist_bonus_culture_effect = { CULTURE = ling_cul_1 }
		linguist_bonus_culture_effect = { CULTURE = ling_cul_2 }
		linguist_bonus_culture_effect = { CULTURE = ling_cul_3 }
		#### Create Linguists
		hidden_effect = {
			# Poor Linguist
			if = {
				limit = { has_trait = greedy }
				create_character = {
					dynasty = none
					template = court_linguist_arabic_cheap_template
					location = root.capital_province
					save_scope_as = cheap_linguist
				}
			}
			# Average Linguist
			create_character = {
				dynasty = none
				template = court_linguist_arabic_medium_template
				location = root.capital_province
				save_scope_as = medium_linguist
			}
			# Good Linguist
			create_character = {
				dynasty = none
				template = court_linguist_arabic_good_template
				culture = scope:ling_cul_2
				faith = scope:ling_cul_2_ruler.faith
				location = root.capital_province
				save_scope_as = good_linguist
			}
			# Scholarly Linguist
			if = {
				limit = { has_trait = scholar }
				create_character = {
					dynasty = none
					template = court_linguist_arabic_good_template
					culture = scope:ling_cul_1
					faith = scope:ling_cul_1_ruler.faith
					location = root.capital_province
					trait = scholar
					save_scope_as = scholarly_linguist
				}
				if = {
					limit = {
						exists = scope:ling_cul_2_bonus
						NOT = {
							scope:scholarly_linguist = { knows_language_of_culture = scope:ling_cul_2_bonus }
						}
					}
					scope:scholarly_linguist = { learn_language_of_culture = scope:ling_cul_2_bonus }
				}
			}
		}
	}

	# Expensive linguist, great teacher
	option = {
		name = me_religious_decisions.0007.a
		pay_short_term_gold = {
			target = scope:good_linguist
			gold = medium_gold_value
		}
		learn_language_linguist_arabic_effect = { LINGUIST = scope:good_linguist }
		stress_impact = {
			greedy = medium_stress_impact_gain
		}
		ai_chance = {
			base = 100
			modifier = {
				short_term_gold < medium_gold_value
				factor = 0
			}
		}
	}
	
	# Cheaper linguist; less learning, not a good teacher
	option = {
		name = me_religious_decisions.0007.b
		pay_short_term_gold = {
			target = scope:medium_linguist
			gold = minor_gold_value
		}
		learn_language_linguist_arabic_effect = { LINGUIST = scope:medium_linguist }
		stress_impact = {
			greedy = medium_stress_impact_gain
		}
		ai_chance = {
			base = 100
			modifier = {
				short_term_gold < minor_gold_value_check
				factor = 0
			}
		}
	}
	
	# I know a good tutor through scholarly circles!
	option = {
		name = me_religious_decisions.0007.c
		trait = scholar
		trigger = { has_trait = scholar }
		learn_language_linguist_arabic_effect = { LINGUIST = scope:scholarly_linguist }
		ai_chance = {
			base = 100
		}
	}
	
	# I'm cheap, get me anyone who can vaguely speak my language
	option = {
		name = me_religious_decisions.0007.d
		trait = greedy
		trigger = { has_trait = greedy }
		learn_language_linguist_arabic_effect = { LINGUIST = scope:cheap_linguist }
		ai_chance = {
			base = 1000
			modifier = { # Don't do it if you can hire the cool guy
				has_trait = scholar
				factor = 0
			}
		}
	}
	
	# On seconds thoughts, I won't hire a linguist at all...
	option = {
		name = me_religious_decisions.0007.e
		ai_chance = {
			base = 0
			modifier = {
				short_term_gold < minor_gold_value_check
				NOR = {
					has_trait = scholar
					has_trait = greedy
				}
				add = 10
			}
		}
	}

	after = {
		every_in_list = {
			list = court_linguist_list
			limit = {
				NOT = { is_courtier_of = root }
			}
			silent_disappearance_effect = yes
		}
	}
