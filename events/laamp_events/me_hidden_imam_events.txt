## Adventurer Inheritance Event
me_hidden_imam_event.0001 = {
	type = character_event
	title = me_hidden_imam_event.0001.t
	desc = me_hidden_imam_event.0001.desc
	theme = laamp
	left_portrait = {
		character = scope:previous_holder

		triggered_animation = {
			trigger = { scope:previous_holder = { is_alive = no } }
			animation = dead
		}
	}
	right_portrait = {
		character = root
		animation = zealous
	}

	trigger = {
		government_has_flag = government_is_landless_adventurer
		faith = faith:ismaili
		OR = {
			has_trait = sayyid
			has_trait = ubaydi
		}
		scope:previous_holder = {
			has_trait = me_hidden_imam
			faith = faith:ismaili
		}
	}

	immediate = {
	}

	option = {
		name = me_hidden_imam_event.0001.a
		
		add_trait = me_hidden_imam
		add_trait = ME_missionary
		if = {
			limit = { scope:previous_holder = { has_trait_xp = { trait = me_hidden_imam value >= 100 } } }
			add_trait_xp = { trait = me_hidden_imam value = 100 }
		}
		else_if = {
			limit = { scope:previous_holder = { has_trait_xp = { trait = me_hidden_imam value >= 50 } } }
			add_trait_xp = { trait = me_hidden_imam value = 50 }
		}
		
		if = {
			limit = {
				has_trait_xp = {
					trait = me_hidden_imam
					value >= 50
				}
				NOT = { has_variable = hidden_imam_can_become_mahdi }
			}
			set_variable = { name = hidden_imam_can_become_mahdi value = yes }
		}
		if = {
			limit = {
				has_trait_xp = {
					trait = me_hidden_imam
					value >= 100
				}
				NOT = { has_variable = hidden_imam_can_reform_faith }
			}
			set_variable = { name = hidden_imam_can_reform_faith value = yes }
		}
	}
}

## Hidden Imam gains land and becomes Caliph
me_hidden_imam_event.0002 = {
	type = character_event
	title = me_hidden_imam_event.0002.t
	desc = me_hidden_imam_event.0002.desc
	theme = faith
	left_portrait = {
		character = scope:reformer
		animation = personality_bold
	}

	trigger = {
		NOT = { has_character_flag = became_ismaili_caliph_flag }
		NOT = { government_has_flag = government_is_landless_adventurer }
		faith = faith:ismaili
		is_male = yes
		OR = {
			has_trait_xp = {
				trait = me_hidden_imam
				value >= 50
			}
			has_trait = ME_ismaili_mahdi
		}
		OR = {
			has_trait = sayyid
			has_trait = ubaydi
		}
	}

	immediate = {
		add_character_flag = became_ismaili_caliph_flag
		play_music_cue = "mx_cue_faith_conversion"
		character:33917 = { save_scope_as = maktum }
		me_create_ismaili_caliphate_scripted_effect = yes
	}

	option = {
		name = me_hidden_imam_event.0002.a
		
		every_player = {
			limit = {
				NOT = { this = root }
				faith = { religion_tag = islam_religion }
			}
			trigger_event = me_hidden_imam_event.0003 # Notification.
		}
	}
	option = {
		name = me_hidden_imam_event.0002.b
		trigger = { has_character_flag = can_become_ismaili_mahdi_flag }
		custom_tooltip = me_hidden_imam_event.0002.b.tt
		
		show_as_tooltip = { ME_become_ismaili_mahdi_decision_effect = yes }
		trigger_event = me_hidden_imam_event.1001
	}
	option = {
		name = me_hidden_imam_event.0002.c
		trigger = { has_trait = ME_ismaili_mahdi }
		custom_tooltip = me_hidden_imam_event.0002.c.tt
		
		trigger_event = me_hidden_imam_event.1003
	}
}
