			desc = me_missionary_event.0001.desc.dead
		}

			trigger = { scope:previous_holder = { is_alive = yes } }
			desc = me_missionary_event.0001.desc.alive
		}
	}
	theme = laamp
	left_portrait = {
		character = scope:previous_holder

		triggered_animation = {
			trigger = { scope:previous_holder = { is_alive = no } }
			animation = dead
		}
	}
	right_portrait = {
		character = root
		animation = zealous
	}

	trigger = {
		government_has_flag = government_is_landless_adventurer
		NOT = { has_trait = me_hidden_imam }
		OR = {
			faith = scope:previous_holder.faith
			root.secret_faith = scope:previous_holder.secret_faith
		}
		scope:previous_holder = {
			has_trait = ME_missionary
		}
	}

	immediate = {
	}

	option = {
		name = me_missionary_event.0001.a
		stress_impact = {
			cynical = major_stress_impact_gain
		}
		
		add_trait = ME_missionary
	}
	option = {
		name = me_missionary_event.0001.b
		stress_impact = {
			zealous = major_stress_impact_gain
		}
	}
}
