# Young character memorizes the Qur'an *really* fast
become_hafidh_ongoing.1000 = {
	type = character_event
	title = become_hafidh_ongoing.1000.t
	desc = become_hafidh_ongoing.1000.desc
	theme = education
	left_portrait = {
		character = root
		animation = happiness
	}
	widget = {
		gui = "event_window_widget_scheme"
		container = "custom_widgets_container"
	}
	
	trigger = {
		NOT = { has_variable = had_event_become_hafidh_ongoing_1000 }
		age <= 20
		is_available = yes
		ai_energy > -1
		is_travelling = no
	}

	weight_multiplier = {
		base = 1
		modifier = {
			add = 10
			OR = {
				has_trait = zealous
				ai_zeal >= 50
			}
		}
		modifier = {
			add = 15
			has_trait = diligent
		}
		modifier = {
			add = 15
			is_adult = no
		}
	}

	immediate = {
		set_variable = {	
			name = had_event_become_hafidh_ongoing_1000
			value = yes
			days = 1460
		}
	}

	option = { # Yes, spend all my focus on this!
		name = become_hafidh_ongoing.1000.a
		
		custom_tooltip = scheme_executes_immediately
		hidden_effect = {
			scope:scheme = {
				add_scheme_progress = 10
			}
		}
		
		stress_impact = {
			impatient = massive_stress_impact_gain
			lazy = massive_stress_impact_gain
			rowdy = medium_stress_impact_gain
			fickle = minor_stress_impact_gain
		}
		ai_chance = {
			base = 100
			modifier = {
				factor = 0.1
				OR = {
					has_trait = lazy
					has_trait = impatient
					ai_energy <= -25
				}
			}
		}
	}

	option = { # Nah... I have better things to do
		name = become_hafidh_ongoing.1000.b
		
		stress_impact = {
			diligent = massive_stress_impact_gain
			zealous = major_stress_impact_gain
			ambitious = medium_stress_impact_gain
			patient = medium_stress_impact_gain
			humble = minor_stress_impact_gain
		}
		ai_chance = {
			base = 10
		}
	}
}

# Your tutor is a heretic, and interprets the verses in an unorthodox way
become_hafidh_ongoing.1020 = {
	type = character_event
	title = become_hafidh_ongoing.1020.t
	desc = become_hafidh_ongoing.1020.desc
	theme = education
	left_portrait = {
		character = root
		animation = personality_rational
	}
	right_portrait = {
		character = scope:target
		animation = chancellor
	}
	widget = {
		gui = "event_window_widget_scheme"
		container = "custom_widgets_container"
	}
	
	trigger = {
		NOT = { has_variable = had_event_become_hafidh_ongoing_1020 }
		age >= 16
		scope:target.faith = {
			faith_hostility_level = {
				target = root.faith
				value >= faith_astray_level
			}
		}
		is_available = yes
		is_travelling = no
	}

	weight_multiplier = {
		base = 1
		modifier = {
			add = 10
			OR = {
				has_trait = zealous
				ai_zeal >= 50
			}
		}
		modifier = {
			add = 15
			has_trait = scholar
		}
		modifier = {
			add = 15
			has_trait = theologian
		}
		modifier = {
			add = 15
			learning >= high_skill_rating # 15
		}
	}

	immediate = {
		set_variable = {	
			name = had_event_become_hafidh_ongoing_1020
			value = yes
			days = 1460
		}
	}

	option = { # Learning challenge against heretic
		name = become_hafidh_ongoing.1020.a
		
		duel = {
			skills = { diplomacy learning }
			target = scope:target
			50 = { # You convert them!
				desc = become_hafidh_ongoing.1020.a.conversion_succesful.desc
				compare_modifier = {
					value = scope:duel_value
					multiplier = 0.5
				}
				modifier = {
					has_perk = zealous_proselytizer_perk
					add = 25
				}
				modifier = {
					has_trait = zealous
					add = 25
				}
				send_interface_toast = {
					title = become_hafidh_ongoing.1020.a.conversion_succesful.t
					left_icon = root
					right_icon = scope:target
					scope:target = {
						set_character_faith = root.faith
					}
					add_piety = medium_piety_gain
				}
				progress_towards_friend_effect = {
					REASON = friend_religious_revalation
					CHARACTER = scope:target
					OPINION = default_friend_opinion
				}
			}
			50 = { # You fail
				desc = become_hafidh_ongoing.1020.a.conversion_failed.desc
				compare_modifier = {
					value = scope:duel_value
					multiplier = -0.5
				}
				send_interface_toast = {
					title = become_hafidh_ongoing.1020.a.conversion_failed.t
					left_icon = root
					right_icon = scope:target
					add_piety = minor_piety_loss
				}
			}
		}
		
		ai_chance = {
			base = 10
			ai_value_modifier = {
				ai_zeal = 1
				ai_boldness = 0.5
			}
		}

	}
	
	option = {
		name = become_hafidh_ongoing.1020.b
		
		set_character_faith = scope:target.faith
		scope:target = {
			progress_towards_friend_effect = {
				REASON = friend_religious_revalation
				CHARACTER = root
				OPINION = default_friend_opinion
			}
		}
		stress_impact = {
			zealous = massive_stress_impact_gain
		}
		ai_chance = {
			base = 0
		}
	}
	
	option = {
		name = become_hafidh_ongoing.1020.c
		
		custom_tooltip = scheme_gains_some_progress
		hidden_effect = {
			scope:scheme = {
				add_scheme_progress = 3
			}
		}
		stress_impact = {
			zealous = minor_stress_impact_gain
		}
		ai_chance = {
			base = 10
		}
	}
}


# Character is unable to pronounce the words of the Qur'an properly - main event
become_hafidh_ongoing.1030 = {
	hidden = yes
	
	
	trigger = { 
		NOT = { has_variable = had_event_become_hafidh_ongoing_1030 }
		NOT = { culture = { has_cultural_pillar = language_arabic } }
		age >= 16
		is_available = yes
		is_travelling = no
	}	
	
	immediate = {	
		if = {
			limit = { scope:target = { has_trait = impatient } }
			trigger_event = become_hafidh_ongoing.1031
		}
		else = { trigger_event = become_hafidh_ongoing.1032 }
	}
}

# Impatient tutor insults you for your inability to pronouce words correctly
become_hafidh_ongoing.1031 = {
	type = character_event
	title = become_hafidh_ongoing.1031.t
	desc = become_hafidh_ongoing.1031.desc
	theme = education
	left_portrait = {
		character = root
		animation = disbelief
	}
	right_portrait = {
		character = scope:target
		animation = anger
	}
	widget = {
		gui = "event_window_widget_scheme"
		container = "custom_widgets_container"
	}
	
	trigger = {
	}

	immediate = {
		set_variable = {	
			name = had_event_become_hafidh_ongoing_1030
			value = yes
			days = 1460
		}
	}

	option = { # Attempt to continue your studies
		name = become_hafidh_ongoing.1031.a
		
		stress_impact = {
			vengeful = medium_stress_impact_gain
			wrathful = medium_stress_impact_gain
		}
		random_list = { 
			50 = { # Your attempt fails, you are set back in your progress
				custom_tooltip = scheme_loses_some_progress
				hidden_effect = {
					scope:scheme = {
						add_scheme_progress = -3
					}
				}
			}
			10 = { # Your attempt succeeds
				custom_tooltip = scheme_gains_some_progress
				hidden_effect = {
					scope:scheme = {
						add_scheme_progress = 3
					}
				}
			}
		}
		ai_chance = {
			base = 100
			modifier = {
				factor = 0.1
				OR = {
					has_trait = vengeful
					has_trait = wrathful
					ai_compassion <= -25
				}
			}
		}

	}
	
	option = {
		name = become_hafidh_ongoing.1031.b
		
		custom_tooltip = scheme_loses_some_progress
		hidden_effect = {
			scope:scheme = {
				add_scheme_progress = -3
			}
		}
		stress_impact = {
			forgiving = medium_stress_impact_gain
			craven = minor_stress_impact_gain
			shy = minor_stress_impact_gain
			patient = minor_stress_impact_gain
			calm = minor_stress_impact_gain
		}
		duel = { # Attempt to intimidate your tutor
			skills = { intrigue prowess }
			target = scope:nervous_servant
			50 = { # You intimidate them
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
				}
				send_interface_toast = {
					title = become_hafidh_ongoing.1031.intimidation_worked
					left_icon = scope:target
					right_icon = root
					scope:secret_to_discover = {
						reveal_to = root
					}
				}
			}
			50 = { # You fail
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
				}
				send_interface_toast = {
					title = become_hafidh_ongoing.1031.intimidation_worked
					left_icon = scope:target
					right_icon = root
					add_opinion = {
						target = root
						opinion = -10
						modifier = me_stuttering_buffoon_opinion
					}
				}
			}
		}

		ai_chance = {
			base = 100
		}
	}
	option = { # Execute your tutor
		name = become_hafidh_ongoing.1031.c
		
		trigger = {
			OR = {
				has_trait = callous
				has_trait = sadistic
				has_trait = arbitrary
			}
		}

		stress_impact = {
			forgiving = major_stress_impact_gain
			compassionate = major_stress_impact_gain
			just = medium_stress_impact_gain
			patient = minor_stress_impact_gain
			calm = minor_stress_impact_gain
			sadistic = medium_stress_impact_loss
			vengeful = medium_stress_impact_loss
			arrogant = minor_stress_impact_loss
		}
		execute_prisoner_effect = {
			VICTIM = scope:target
			EXECUTIONER = root
		}

		ai_chance = {
			base = 100
			modifier = {
				factor = 0.1
				OR = {
					has_trait = forgiving
					has_trait = just
					has_trait = compassionate
					ai_compassion >= 10
				}
			}
		}

	}
}
