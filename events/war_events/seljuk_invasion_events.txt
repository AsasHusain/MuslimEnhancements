namespace = seljuk_invasion_events

#The Seljuk Emperor recieves news of Suleyman's takeover
seljuk_invasion_events.0001 = {
		
	title = seljuk_invasion_events.0001.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					character:danishmendid_1 = {
						is_alive = yes
					}
				}
				desc = seljuk_invasion_events.0001.desc
			}
			desc = seljuk_invasion_events.0001.desc_danishdead
		}
	}
	theme = war
	
	# Left portrait is Suleyman, right portrait is Seljuk Emperor
	left_portrait = {
		character = character:3046
		animation = personality_dishonorable
	}
	
	right_portrait = {
		character = title:e_persia.holder
		animation = war_over_win
	}
	
	lower_right_portrait = { # Lower right is Danishmend
		character = character:danishmendid_1
	}

	immediate = {
		custom_tooltip = seljuk_rum_takeover_sulayman
		if = {
			limit = {
				character:danishmendid_1 = {
					is_alive = yes
				}
			}
			custom_tooltip = seljuk_rum_takeover_danishmend
		}
		character:3046 = { save_scope_as = sulayman }
		character:danishmendid_1 = { save_scope_as = danishmend }
	}

	option = {
		name = seljuk_invasion_events.0001.a
		set_player_character = character:3046
		ai_chance = {
			base = 0
		}
	}

	option = {
		name = seljuk_invasion_events.0001.c
		set_player_character = character:danishmendid_1
		trigger = {
			character:danishmendid_1 = {
				is_alive = yes
			}
		}
		ai_chance = {
			base = 0
		}
	}

	option = {
		name = seljuk_invasion_events.0001.b
		
		ai_chance = {
			base = 100
		}
	}
}

# Sulayman and Danishmend players chose whether to gain land or not
seljuk_invasion_events.0008 = {
		
	title = seljuk_invasion_events.0008.title
	desc = seljuk_invasion_events.0008.desc
	theme = war
	
	right_portrait = {
		character = root
		animation = war_over_win
	}
	
	immediate = {
	}
	
	option = { # Become a landed ruler
		name = seljuk_invasion_events.0008.a
		if = {
			limit = { this = character:3046 } # Sulayman ibn Qutalmish
			hidden_effect = { me_seljuk_invasion_sulayman_effect = yes }
			custom_tooltip = seljuk_invasion_events.0008.tt.sulayman
		}
		if = {
			limit = { this = character:danishmendid_1 } # Danishmend
			hidden_effect = { me_seljuk_invasion_danishmend_effect = yes }
			custom_tooltip = seljuk_invasion_events.0008.tt.danishmend
		}
	}
	
	option = { # Continue as adventerer
		name = seljuk_invasion_events.0008.b
		custom_tooltip = seljuk_invasion_events.0008.tt.landless
	}
}
