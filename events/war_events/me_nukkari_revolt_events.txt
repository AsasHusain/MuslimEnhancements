### Early Encounter with Abu Yazid in Tozeur, 7 years before the revolt 
me_nukkari_revolt.0001 = {
	title = me_nukkari_revolt.0001.t
	desc = me_nukkari_revolt.0001.desc
	theme = intrigue
	override_background = { reference = bp1_courtyard_mena }

	trigger = {
		OR = { # No one that has control over the county is Nukkari
			NOT = { scope:troubled_town.holder.faith = faith:nukkari }
			NOT = { scope:troubled_town.holder = { any_liege_or_above = { faith = faith:nukkari } } }
		}
	}

	immediate = {

	}

	option = {
		name = me_nukkari_revolt.0001.a
		custom_tooltip = me_nukkari_revolt.0001.a.tt

		random_list = {
			35 = { # Abu Yazid fails to do anything - 0 strength

				custom_tooltip = me_nukkari_revolt.0001.tt.nukkari_fail
				hidden_effect = {
					send_interface_toast = {
						title = me_nukkari_revolt.0001.tt.nukkari_fail
						custom_tooltip = me_nukkari_revolt.0001.tt.likely
						left_icon = scope:troubled_town
					}
				}
			}
			60 = { # Abu Yazid gains some followers - +1 strength 

				custom_tooltip = me_nukkari_revolt.0001.tt.nukkari_mid
				hidden_effect = {
					send_interface_toast = {
						title = me_nukkari_revolt.0001.tt.nukkari_mid
						custom_tooltip = me_nukkari_revolt.0001.tt.likely
						left_icon = scope:troubled_town 
					}
				}

				change_global_variable = {
					name = nukkari_revolt_strength_score
					add = 1
				}
			}
			5 = { # Abu Yazid gains alot of followers - +2 strength

				custom_tooltip = me_nukkari_revolt.0001.tt.nukkari_success
				send_interface_toast = {
					title = me_nukkari_revolt.0001.tt.nukkari_success
					left_icon = scope:troubled_town
					scope:troubled_town = {
						add_county_modifier = {
							modifier = me_nukkari_rebellious
							years = 5
						}
					}
					if = {
						limit = { scope:troubled_town.faith = { has_doctrine = special_doctrine_is_ibadi_faith } }
						scope:troubled_town = { set_county_faith = faith:nukkari }
					}
				}

				change_global_variable = {
					name = nukkari_revolt_strength_score
					add = 2
				}
			}
		}
	}

	option = {
		name = me_nukkari_revolt.0001.b
		custom_tooltip = me_nukkari_revolt.0001.b.tt

		scope:troubled_town = {
			add_county_modifier = {
				modifier = me_nukkari_discontent
				years = 5
			}
		}

		duel = {
			skill = intrigue
			value = high_skill_rating

			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -49
				}
				custom_tooltip = me_nukkari_revolt.0001.tt.prison_fail
				hidden_effect = {
					send_interface_toast = {
						title = me_nukkari_revolt.0001.tt.prison
						custom_tooltip = me_nukkari_revolt.0001.tt.unlikely
					}
				}
				trigger_event = {
					id = me_nukkari_revolt.0002 
					days = 14
				}
			}

			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -49
				}
				custom_tooltip = me_nukkari_revolt.0001.tt.prison_flee
				hidden_effect = {
					send_interface_toast = {
						title = me_nukkari_revolt.0001.tt.prison
						custom_tooltip = me_nukkari_revolt.0001.tt.unlikely
					}
				}
				trigger_event = {
					id = me_nukkari_revolt.0003 
					days = 14
				}
			}
		}
	}

me_nukkari_revolt.0100 = {
	scope = none
	hidden = yes

	immediate = {
		spawn_nukkari_revolt_character_effect = yes
		scope:nukkari_leader = {
			save_scope_as = story_owner
			create_story = story_nukkari_revolt
			copy_inheritable_appearance_from = character:me_ifrinid_001_dna_clone
			give_nickname = nick_sahib_al_himar # He recieved his donkey in 944, but he's more well known by this name so lets move things along
			set_relation_friend = scope:nukkari_al_ama
		}
		scope:nukkari_al_ama = { give_nickname = nick_al_ama }

		every_living_character = {
			limit = { 
				OR = { 
					capital_province = { geographical_region = ghw_region_north_africa } 
					AND = {
						is_ai = no
						any_county_in_region = {
							region = custom_roman_africa
							count >= 1
							holder = root
						}
					}
				}
			}
			trigger_event = {
				id = me_nukkari_revolt.0101
			}
		}
		
		#Random courtiers start joining Abu Yazid's revolt

			limit = {
				NOT = { faith = faith:nukkari }
				is_available = yes
				capital_province = { geographical_region = ghw_region_north_africa }
				NOT = { is_in_list = NR_102_rulers }
			}
			add_to_list = NR_102_rulers
			random_courtier = {
				limit = {
					is_female = no
					faith = root.faith
					NOT = { has_trait = cynical }
					NOT = { is_vassal_of = root }
					is_capable_adult_ai = yes
					is_imprisoned = no
					is_healthy = yes 
				}
				save_scope_as = nr_heretic_courtier
				faith = { save_scope_as = old_faith }
			}
			trigger_event = {
				id = me_nukkari_revolt.0102
				days = { 70 365 }
			}
		}
		random_ruler = {
			limit = {
				NOT = { faith = faith:nukkari }
				is_available = yes
				capital_province = { geographical_region = ghw_region_north_africa }
				NOT = { is_in_list = NR_102_rulers }
			}
			add_to_list = NR_102_rulers
			random_courtier = {
				limit = {
					is_female = no
					faith = root.faith
					NOT = { has_trait = cynical }
					NOT = { is_vassal_of = root }
					is_capable_adult_ai = yes
					is_imprisoned = no
					is_healthy = yes 
				}
				save_scope_as = nr_heretic_courtier
				faith = { save_scope_as = old_faith }
			}
			trigger_event = {
				id = me_nukkari_revolt.0102
				days = { 70 365 }
			}
		}
		random_ruler = {
			limit = {
				NOT = { faith = faith:nukkari }
				is_available = yes
				capital_province = { geographical_region = ghw_region_north_africa }
				NOT = { is_in_list = NR_102_rulers }
			}
			add_to_list = NR_102_rulers
			random_courtier = {
				limit = {
					is_female = no

					NOT = { has_trait = cynical }
					NOT = { is_vassal_of = root }
					is_capable_adult_ai = yes
