### Ha-Mimists
# Warning for Ha-Mimist Invasion
me_berberist_events.0001 = {
	title = me_berberist_events.0001.t
	desc = me_berberist_events.0001.desc
	theme = realm
	override_background = { reference = terrain }
	
	right_portrait = {
		character = root
		animation = personality_bold
	}

	trigger = {
	}

	immediate = {
		faith:hamimist = { save_scope_as = hamimist }
	}

	option = {
		name = me_berberist_events.0001.a
		custom_tooltip = me_berberist_events.0001.a.tt
	}
}

# Setup event for Ha-Mimist Invasion
me_berberist_events.0002 = {
	hidden = yes
	

	trigger = {
	}

	immediate = {
		spawn_hamimist_invasion_character_effect = yes
		me_start_hamimist_invasion_effect = {
			TARGET_COUNTY = title:c_ceuta
			REBEL_LEADER = scope:hamimist_leader
		}
		set_global_variable = {
			name = hamimist_invasion_ended
			value = yes
		}
	}
}

# Ha-Mimist defeat event
me_berberist_events.0004 = {
	title = me_berberist_events.0004.t
	desc = me_berberist_events.0004.desc
	theme = prison

	left_portrait = {
		character = scope:attacker
		animation = prisondungeon
	}

	trigger = {
	}

	immediate = {
	}

	option = {
		name = me_berberist_events.0004.a
		custom_tooltip = me_berberist_events.0004.a.tt
		
		execute_prisoner_effect = {
			VICTIM = scope:attacker
			EXECUTIONER = root
		}
		add_piety = 1000

		hidden_effect = {
			every_prisoner = {
				limit = { faith = faith:hamimist }
				death = {
					death_reason = death_execution
					killer = root
				}
			}
		}
		ai_chance = {
			base = 100
		}
	}
	option = {
		name = me_berberist_events.0004.b
		
		execute_prisoner_effect = {
			VICTIM = scope:attacker
			EXECUTIONER = root
		}
		add_piety = 750
		ai_chance = {
			base = 0
		}
	}
	option = {
		name = me_berberist_events.0004.c
		ai_chance = {
			base = 0
		}
	}
}
