## Italy-Bari Startup
ss_wars.0200 = {
	title = ss_wars.0200.t
	desc = {
		desc = ss_wars.0200.desc
		desc = {
			first_valid = {
				triggered_desc = {
					trigger = { has_global_variable = me_bari_partition_true_variable }
					desc = ss_wars.0200.desc.true
				}
				desc = ss_wars.0200.desc.false
			}
		}
		desc = ss_wars.0200.desc.start
		desc = {
			first_valid = {
				triggered_desc = {
					trigger = { has_game_rule = me_calabria_bari }
					desc = ss_wars.0200.desc.calabria_bari
				}
				desc = ss_wars.0200.desc.calabria_norm
			}
		}
	}
	theme = realm
	override_background = { reference = throne_room }

	trigger = { current_date = 867.1.1 }
	
	left_portrait = {
		character = root
		animation = title:c_bari.holder
	}
	
	option = { # Activate Bari Partition
		name = ss_wars.0200.a

		hidden_effect = {
			set_global_variable = {
				name = me_bari_partition_true_variable
				value = yes
			}
		}
	}
	option = { # Deactivate Bari Partition
		name = ss_wars.0200.b

		remove_global_variable = me_bari_partition_true_variable
	}
}
