namespace = me_alid_egypt_revolt


### Ibn as-Sufi begins his revolt
me_alid_egypt_revolt.0001 = {
	title = me_alid_egypt_revolt.0001.t
	desc = {
		triggered_desc = {
			trigger = { character:me_persian_007 = { is_ai = yes } }
			desc = me_alid_egypt_revolt.0001.desc.ibn_yazdad_dead
		}
		triggered_desc = {
			trigger = { character:me_persian_007 = { is_ai = no } }
			desc = me_alid_egypt_revolt.0001.desc.ibn_yazdad_alive
		}
	}
	theme = realm
	override_background = { reference = terrain }
	
	right_portrait = {
		character = scope:ibn_as_sufi
		animation = assassin
	}
	
	left_portrait = {
		character = scope:ibn_yazdad
		animation = fear
	}

	trigger = {
		NOT = { exists = global_var:ibn_as_sufi_revolt_ended }
	}

	immediate = {
		set_global_variable = { name = ibn_as_sufi_revolt_ended value = yes }
		if = {
			limit = { character:me_hashimid_046 = { is_alive = yes } }
			character:me_hashimid_046 = { save_scope_as = ibn_as_sufi }
		}
		else_if = {
			limit = { character:me_hashimid_046 = { is_alive = no } }
			character:RICE_upper_egypt_013 = { save_scope_as = ibn_as_sufi }
			create_dynamic_title = {
				tier = duchy
				name = d_laamp_me_ibn_as_sufi_867
			}
			create_title_and_vassal_change = {
				type = created
				save_scope_as = change
				add_claim_on_loss = no
			}
			scope:new_title = {
				set_capital_county = title:c_qus
				set_landless_title = yes
				set_destroy_on_gain_same_tier = yes
				set_no_automatic_claims = yes
				set_can_be_named_after_dynasty = no
				change_title_holder = {
					holder = scope:ibn_as_sufi
					change = scope:change
				}
			}
			resolve_title_and_vassal_change = scope:change
			scope:new_title = {
				generate_coa = yes
				set_variable = {
					name = temporary_title
					value = yes
				}
			}
		}
		if = {
			limit = { character:me_persian_007 = { is_ai = yes } }
			character:me_persian_007 = {
				save_scope_as = ibn_yazdad
				death = {
					death_reason = death_me_mutilated_and_crucified
					killer = scope:ibn_as_sufi
				}
			}
		}
		scope:ibn_as_sufi = {
			start_war = {
				cb = me_alid_egypt_revolt_cb
				target = title:k_egypt.holder
				target_title = title:d_al-said
			}
			spawn_army = {
				levies = 700
				men_at_arms = {
					type = mubarizun
					stacks = 2
				}
				men_at_arms = {
					type = light_horsemen
					stacks = 1
				}
				men_at_arms = {
					type = pikemen_unit
					stacks = 1
				}
				men_at_arms = {
					type = bowmen
					stacks = 2
				}
				men_at_arms = {
					type = light_footmen
					stacks = 2
				}
				location = capital_province
				inheritable = yes
				name = me_alid_egyptian_rebel_troops
			}
		}
	}

	option = {
		name = me_alid_egypt_revolt.0001.a
	}
}
