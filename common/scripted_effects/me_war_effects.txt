nukkari_event_launcher_effect = {
	if = { # Early Encounter
		limit = {
			current_date >= 937.1.1
			current_date < 938.1.1
			NOT = { exists = global_var:nukkari_revolt_ended }
		}
		
		set_global_variable = {
			name = nukkari_revolt_strength_score
			value = 5
		}
		random_list = { # Abu Yazid picks a random town in the region he will revolt in (Historically he does this in Tozeur)
		
			5 = {
				title:c_tunis = { save_scope_as = troubled_town }
			}
			5 = {
				title:c_satfura = { save_scope_as = troubled_town }
			}
			5 = {
				title:c_medjerda = { save_scope_as = troubled_town }
			}
			5 = {
				title:c_mahdiya = { save_scope_as = troubled_town }
			}
			5 = {
				title:c_west_qamuda = { save_scope_as = troubled_town }
			}
			5 = {
				title:c_tabassa = { save_scope_as = troubled_town }
			}
		}
		
		if = { # We prioritize the player in this event
			limit = {
				scope:troubled_town.holder = { is_ai = no }
			}
			scope:troubled_town.holder = {
				trigger_event = {
					id = me_nukkari_revolt.0001
					days = { 70 300 }
				}
			}
		}
		else_if = {
			limit = { # Player doesn't control the county
				scope:troubled_town.holder = { is_ai = yes }
				scope:troubled_town.holder = { any_liege_or_above = { is_ai = yes } }
			}
			scope:troubled_town.holder = {
				trigger_event = {
					id = me_nukkari_revolt.0001
					days = { 70 300 }
				}
			}
		}
		else_if = {
			limit = { # Player is county's first liege (Duchy)
				scope:troubled_town.holder.liege = {is_ai = no }
			}
			scope:troubled_town.holder.liege = {
				trigger_event = {
					id = me_nukkari_revolt.0001
					days = { 70 300 }
				}
			}
		}
		else_if = {
			limit = { # Player is county's second liege (Kingdom)
				scope:troubled_town.holder.liege.liege = {is_ai = no }
			}
			scope:troubled_town.holder.liege.liege = {
				trigger_event = {
					id = me_nukkari_revolt.0001
					days = { 70 300 }
				}
			}
		}
		else_if = {
			limit = { # Player is county's third liege (Empire)
				scope:troubled_town.holder.liege.liege.liege = {is_ai = no }
			}
			scope:troubled_town.holder.liege.liege.liege = {
				trigger_event = {
					id = me_nukkari_revolt.0001
					days = { 70 300 }
				}
			}
		}
	}

	if = { # Ibadi/Wahbi counties start converting to Nukkari
		limit = {
			current_date >= 937.1.1
			current_date <= 945.1.1
			NOT = { exists = global_var:nukkari_revolt_ended }
		}
		random_county_in_region = {
			region = ghw_region_north_africa
			limit = { 
				faith = { has_doctrine = special_doctrine_is_ibadi_faith }
				NOT = { is_in_list = NR_004_counties }
			}
			add_to_list = NR_004_counties
			save_scope_as = nr_converted_county
			trigger_event = {
				id = me_nukkari_revolt.0004
				days = { 5 360 }
			}
		}
		if = { # If the Wahbi Imamate doesn't exist, double the amount of counties that convert
			limit = { NOT = { exists = title:d_wahbiya.holder } }
			random_county_in_region = {
				region = ghw_region_north_africa
				limit = { 
					faith = { has_doctrine = special_doctrine_is_ibadi_faith }
					NOT = { is_in_list = NR_004_counties }
				}
				add_to_list = NR_004_counties
				save_scope_as = nr_converted_county
				trigger_event = {
					id = me_nukkari_revolt.0004
					days = { 5 360 }
				}
			}
		}
		random_ruler = { # Tribal rulers have a chance to convert to Nukkarism
			limit = {
				faith = { has_doctrine = special_doctrine_is_ibadi_faith }
				capital_province.faith = { has_doctrine = special_doctrine_is_ibadi_faith }
				NOT = { is_in_list = NR_005_rulers }
				government_has_flag = government_is_tribal
				OR = {
					capital_province = { geographical_region = world_africa_sahara }
					capital_province = { geographical_region = ghw_region_north_africa }
				}
			}
			random = {
				chance = 20
				trigger_event = {
					id = me_nukkari_revolt.0005
					days = { 5 360 }
				}
				add_to_list = NR_005_rulers
			}
		}
	}

	if = { # Abu Yazid spawns in.
		limit = {
			current_date >= 943.1.1
			current_date < 944.1.1
			NOT = { exists = global_var:nukkari_revolt_ended }
		}
		trigger_event = me_nukkari_revolt.0100
	}
}

me_start_nukkari_revolt_effect = {

	$REBEL_LEADER$ = { save_scope_as = rebel_leader }
	$TARGET_COUNTY$ = { save_scope_as = target_county }
	#Give the nukkari leader a title
	create_dynamic_title = {
		tier = duchy
		name = me_nukkari_revolt_name
	}
	create_title_and_vassal_change = {
		type = created
		save_scope_as = change
		add_claim_on_loss = no
	}
	scope:new_title = {
		set_capital_county = scope:target_county
		set_landless_title = yes
		set_destroy_on_gain_same_tier = yes
		set_no_automatic_claims = yes
		set_can_be_named_after_dynasty = no
		change_title_holder = {
			holder = scope:rebel_leader
			change = scope:change
		}
	}
	resolve_title_and_vassal_change = scope:change
	scope:new_title = {
		generate_coa = yes
		set_variable = {
			name = temporary_title
			value = yes
		}
	}
	#Declare the war(s), every independent ruler gets into a seperate war	
	every_living_character = {
		limit = { 
			NOT = { faith = faith:nukkari }
			OR = {
				has_title = title:c_kairwan
				has_title = title:c_west_qamuda
				has_title = title:c_tabassa
				has_title = title:c_tunis
				has_title = title:c_satfura
				has_title = title:c_medjerda
				has_title = title:c_mahdiya
			}
		}
		top_liege = { add_to_list = NR_defenders }
	}
	every_in_list = {
		list = NR_defenders
		scope:rebel_leader = {
			start_war = {
				cb = me_nukkari_revolt_war
				target = PREV
				target_title = title:d_kairouan
			}
		}
		trigger_event = me_nukkari_revolt.0201
	}
	
	# Setup event troops.
