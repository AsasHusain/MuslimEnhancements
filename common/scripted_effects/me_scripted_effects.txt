me_add_doctrine_startup_effect = {
	if = {
		limit = { # Double-check to make certain this has not already happened
			NOT = { exists = global_var:me_flag_add_doctrine }	
		}
		set_global_variable = { # Add global variable to prevent this from happening more than once
			name = me_flag_add_doctrine
			value = yes
		}
		every_religion_global = {
			every_faith = {
				## Add intoxication crime doctrine to faiths
				if = {
					limit = { 
						NOR = { 
							religion_tag = islam_religion
							religion_tag = jainism_religion
							trait_is_sin = drunkard ## Just incase any modded faith has drunkard as a sin
							this = faith:zurvanism # Behafardism
							this = faith:sabianism
							this = faith:mandeaism 
							this = faith:priscillianism
							this = faith:bogomilist
							has_doctrine = doctrine_me_intoxication_accepted
							has_doctrine = doctrine_me_intoxication_shunned
							has_doctrine = doctrine_me_intoxication_crime
							has_doctrine = doctrine_me_intoxication_virtuous
						} 
					}
					add_doctrine = doctrine_me_intoxication_accepted
				}
				else_if = {
					limit = { 
						OR = {
							trait_is_sin = drunkard ## Just incase any modded faith has drunkard as a sin
							this = faith:sabianism
							this = faith:mandeaism 
							this = faith:priscillianism
							this = faith:bogomilist
							religion_tag = jainism_religion
							
						}
						NOR = {
							has_doctrine = doctrine_me_intoxication_accepted
							has_doctrine = doctrine_me_intoxication_shunned
							has_doctrine = doctrine_me_intoxication_crime
							has_doctrine = doctrine_me_intoxication_virtuous
						}
					}
					add_doctrine = doctrine_me_intoxication_shunned
				}
#				else_if = { #Placeholder incase it is needed to give a faith the crime doctine
#					limit = { 
#						OR = { 
#						} 
#					}
#					add_doctrine = doctrine_me_intoxication_crime
#				}
#				else_if = { #Placeholder incase it is needed to give a faith the virtuous doctine
#					limit = { 
#						OR = { 
#						} 
#					}
#					add_doctrine = doctrine_me_intoxication_virtuous
#				}

				## Add Ablution doctrine to faiths
				if = {
					limit = { 
						NOR = { 
							religion_tag = islam_religion
							religion_tag = judaism_religion
							this = faith:mandeaism
							has_doctrine = doctrine_me_ablution_handwashing
							has_doctrine = doctrine_me_ablution_full
						}
					}
					add_doctrine = doctrine_me_ablution_none
				}
				else_if = { # Jewish Handwashing
					limit = { religion_tag = judaism_religion }
					add_doctrine = doctrine_me_ablution_handwashing
				}
				else_if = { # Mandaean Rishama
					limit = { this = faith:mandeaism }
					add_doctrine = doctrine_me_ablution_full
				}
			}
		}
	}
}

	}

		name = subject_is_set
		value = $SET_SUBJECT$
	}
	save_temporary_scope_value_as = {
		name = topic_is_set
		value = $SET_TOPIC$
	}
	if = {
		limit = {
			exists = scope:dummy_gender
			scope:author = scope:dummy_gender
		}
		clear_saved_scope = author
		clear_saved_scope = dummy_gender
	}
	else = { #If we have an author we save them as a variable on the owner so that we can access them later on when setting the subject etc.
		scope:owner = {
			set_variable = {
				name = book_artifact_author
				value = scope:author
				days = 1
			}
		}
	}
	hidden_effect_new_object = {
		# Get artifact quality and wealth
		get_artifact_quality_effect = yes
		get_artifact_wealth_effect = yes

		get_book_variables_effect = yes

		if = { #Should this be a court artifact?
			limit = {
				NOT = { exists = scope:should_be_trinket }
			}
			if = { #Do we have an author?
				limit = {
					exists = scope:author
				}
				if = { #Is this author using Palm Leaf (indian) books?
					limit = {
						scope:author = { artifact_use_indian_building_visuals_trigger = yes }
					}
					scope:owner = {
						create_artifact = {	
							name = placeholder # Will be regenerated in the post-creation effects
							creator = scope:author
							description = placeholder # Will be regenerated in the post-creation effects

							type = book
							modifier = artifact_placeholder_modifier

							quality = scope:quality

						}
					}
				}
				else = { #Codices/Non-Indian books

						create_artifact = {	
							name = placeholder # Will be regenerated in the post-creation effects
							creator = scope:author
							description = placeholder # Will be regenerated in the post-creation effects
							visuals = book
							type = book
							modifier = artifact_placeholder_modifier
							save_scope_as = newly_created_artifact
							wealth = scope:wealth
							quality = scope:quality
							#decaying = no
						}
					}
				}
			}
			else = { #We don't have an author
				if = {
					limit = { #Is this a palm leaf/indian book?
						scope:owner = { artifact_use_indian_building_visuals_trigger = yes }
					}

					scope:owner = {
						create_artifact = {	
							name = placeholder # Will be regenerated in the post-creation effects
							description = placeholder # Will be regenerated in the post-creation effects
							visuals = book
							type = book
							history = { type = created_before_history } # We don't know who wrote it
