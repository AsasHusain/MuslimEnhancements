### Hijri Calendar Customizable Localization ###

# Main date display switcher: returns Hijri date for Muslim players, Gregorian otherwise
ME_GetDisplayDate = {
	type = character

	text = {
		trigger = {
			has_game_rule = me_hijri_enabled
			faith = { religion_tag = islam_religion }
			exists = global_var:me_hijri_year
		}
		localization_key = ME_HIJRI_DISPLAY_DATE
	}
	text = {
		localization_key = ME_GREGORIAN_DISPLAY_DATE
		fallback = yes
	}
}

# Returns the Hijri month name based on the current global variable
# Triggers are evaluated in order; first match wins
ME_GetHijriMonthName = {
	type = character

	# Month 1: Muharram
	text = {
		trigger = { me_hijri_month_value < 2 }
		localization_key = me_hijri_month_muharram
	}
	# Month 2: Safar
	text = {
		trigger = { me_hijri_month_value < 3 }
		localization_key = me_hijri_month_safar

	# Month 3: Rabi' al-Awwal

		trigger = { me_hijri_month_value < 4 }
		localization_key = me_hijri_month_rabi_al_awwal
	}
	# Month 4: Rabi' al-Thani

		trigger = { me_hijri_month_value < 5 }
		localization_key = me_hijri_month_rabi_al_thani
	}

	text = {
		trigger = { me_hijri_month_value < 6 }
		localization_key = me_hijri_month_jumada_al_ula
	}
	# Month 6: Jumada al-Thani
	text = {
		trigger = { me_hijri_month_value < 7 }
		localization_key = me_hijri_month_jumada_al_thani
	}
	# Month 7: Rajab
	text = {
		trigger = { me_hijri_month_value < 8 }
		localization_key = me_hijri_month_rajab
	}
	# Month 8: Sha'ban
	text = {
		trigger = { me_hijri_month_value < 9 }
		localization_key = me_hijri_month_shaban
	}
	# Month 9: Ramadan
	text = {
		trigger = { me_hijri_month_value < 10 }
		localization_key = me_hijri_month_ramadan
	}
	# Month 10: Shawwal
	text = {
		trigger = { me_hijri_month_value < 11 }
		localization_key = me_hijri_month_shawwal
	}
	# Month 11: Dhu al-Qi'dah
	text = {
		trigger = { me_hijri_month_value < 12 }
		localization_key = me_hijri_month_dhu_al_qidah
	}
	# Month 12: Dhu al-Hijjah (fallback)
	text = {
		localization_key = me_hijri_month_dhu_al_hijjah
		fallback = yes
	}
}
