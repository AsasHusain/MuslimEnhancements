ME_ArabicCaliphateEmpire = {
	type = character
	text = {
		trigger = {
			highest_held_title_tier = tier_empire
			faith = {
				religious_head = root
				has_doctrine = doctrine_temporal_head
				religion_tag = islam_religion
				OR = {
					has_doctrine = muhammad_succession_sunni_doctrine
					root = { has_title = title:d_sunni }
					root = { has_title = title:d_shiite }
				}
			}
		}
		localization_key = empire_arabic_caliphate
	}
	text = {
		fallback = yes
		localization_key = empire_arabic_default
	}
}

			highest_held_title_tier = tier_duchy
		}
		localization_key = duke_male_arabic_hidden_imam
	}
	text = {
		trigger = {
			OR = { # All Hashimids were considered Sharifs, but this will be reserved for Sayyids
				has_trait = sayyid
				has_trait = ubaydi
			}
			NOT = { has_trait = me_hidden_imam }
			highest_held_title_tier = tier_duchy

		localization_key = duke_male_arabic_sharif_landless
	}
	# Mercenary
	text = {
		trigger = { has_realm_law = camp_purpose_mercenaries }
		localization_key = duke_male_arabic_mercenary

	# Wanderer
	text = {
		trigger = { has_realm_law = camp_purpose_wanderers }
		fallback = yes
		localization_key = duke_male_arabic_wanderer
	}

	# Scholar
	text = {
		trigger = { has_realm_law = camp_purpose_scholars }
		localization_key = duke_male_arabic_scholar
	}

	# Explorer
	text = {
		trigger = { has_realm_law = camp_purpose_explorers }
		localization_key = duke_male_arabic_explorer
	}

	# Freebooter
	text = {
		trigger = { has_realm_law = camp_purpose_brigands }
		localization_key = duke_male_arabic_freebooter
	}

	# Legitimist
	text = {
		trigger = { has_realm_law = camp_purpose_legitimists }
		localization_key = duke_male_arabic_legitimists

	text = {

	}
