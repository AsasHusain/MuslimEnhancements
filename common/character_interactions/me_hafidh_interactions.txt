become_hafidh_interaction = {
	icon = icon_scheme_become_hafidh
	category = interaction_category_friendly
	common_interaction = yes

	send_name = START_SCHEME

	interface_priority = 48
	scheme = become_hafidh
	ignores_pending_interaction_block = yes

	ai_targets = {
		ai_recipients = vassals
		ai_recipients = courtiers
	}
	ai_frequency = 48

	is_shown = {
		scope:actor = {
			faith.religion = religion:islam_religion
			NOT = { faith = { has_doctrine = muhammad_succession_berberist_doctrine } }
			NOT = { faith = faith:druze }
			NOT = {
				has_trait_xp = {
					trait = hafidh
					value >= 100
				}
			}
		}
		scope:recipient = {
			faith.religion = religion:islam_religion
			NOT = { faith = { has_doctrine = muhammad_succession_berberist_doctrine } }
			NOT = { faith = faith:druze }
			OR = {
				is_courtier_of = scope:actor
				is_vassal_or_below_of = scope:actor
			}
			is_adult = yes
		}
		NOT = { scope:actor = scope:recipient }
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			can_start_scheme = {
				type = become_hafidh
				target_character = scope:recipient
			}
		}
	}

	desc = scheme_interaction_tt_become_hafidh_approved

	on_accept = {
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = become_hafidh_interaction_notification

				left_icon = scope:actor					
				right_icon = scope:recipient

				begin_scheme_basic_effect = {
					SCHEME_TYPE = become_hafidh
					TARGET_TYPE = target_character
					TARGET_SCOPE = scope:recipient
				}
			}
		}
	}
	
	ai_potential = {
		primary_title.tier >= tier_county
		NOR = {
			has_trait = lazy
			has_trait = cynical
			scheme_generic_ai_blocker_trigger = yes
			primary_title = {
				is_mercenary_company = yes
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 25
			scope:actor = {
				has_trait = zealous
			}
		}
		
		modifier = {
			add = 25
			scope:actor = {
				has_trait = ambitious
			}
		}
		
		modifier = {
			add = 50
			scope:actor = {
				is_adult = no
			}
		}
		
		modifier = {
			add = 25
			scope:recipient = {
				has_relation_friend = scope:actor
			}
		}
		
		modifier = {
			add = 25
			scope:recipient = {
				has_relation_lover = scope:actor
			}
		}
		
		modifier = {
			add = 10
			scope:recipient = {
				has_relation_potential_friend = scope:actor
			}
		}
		
		modifier = {
			add = 5
			scope:recipient = {
				is_close_family_of = scope:actor
				is_ruler = yes
			}
		}
		
		modifier = { # Knows Arabic
			add = 25
			scope:actor = { 
				knows_language = language_arabic 
			}
		}
		
		modifier = { # Tutor knows Arabic
			add = 25
			scope:recipient = { 
				knows_language = language_arabic 
			}
		}
		
		modifier = {
			add = 20
			scope:actor = {
				has_focus = learning_theology_focus
			}
		}
		
		modifier = {
			add = 100
			scope:actor.culture = {
				has_cultural_tradition = tradition_zealous_people
			}
		}
		
		modifier = {
			factor = 2
			scope:actor = {
				learning >= 12
			}
		}
		
		modifier = {
			factor = 0.5
			scope:actor = {
				learning <= 6
			}
		}
	}
}

make_child_become_hafidh_interaction = {
	category = interaction_category_friendly
	common_interaction = yes
	icon = icon_scheme_become_hafidh

	send_name = memorize_quran

	desc = make_child_become_hafidh_interaction_desc

	redirect = {
		scope:actor = {
			if = {
				limit = {
					scope:recipient = {
						can_start_scheme = { type = become_hafidh target_character = prev }
					}
				}
				save_scope_as = secondary_actor
			}
			else_if = {
				limit = {
					any_courtier = {
						scope:recipient = {
							can_start_scheme = { type = become_hafidh target_character = prev }
						}
					}
				}
				random_courtier = {
					limit = {
						scope:recipient = {
							can_start_scheme = { type = become_hafidh target_character = prev }
						}
					}
					save_scope_as = secondary_actor
				}
			}
			else_if = {
				limit = {
					any_vassal = {
						scope:recipient = {
							can_start_scheme = { type = become_hafidh target_character = prev }
						}
					}
				}
				random_vassal = {
					limit = {
						scope:recipient = {
							can_start_scheme = { type = become_hafidh target_character = prev }
						}
					}
					save_scope_as = secondary_actor
				}
			}
		}
		scope:recipient = {
			save_scope_as = secondary_recipient

			if = {
				limit = {
					exists = employer
					NOT = { employer = scope:recipient }
				}
				
				employer = {
					save_scope_as = recipient
				}
			}
		}
	}

	populate_actor_list = {
		scope:actor = {
			if = {
				limit = {
					is_physically_able_adult = yes
					exists = scope:secondary_recipient
					scope:secondary_recipient = {
						can_start_scheme = { type = become_hafidh target_character = prev }
					}
				}
				add_to_list = characters
			}
			every_courtier = {
				limit = {
					is_physically_able_adult = yes
					exists = scope:secondary_recipient
					scope:secondary_recipient = {
						can_start_scheme = { type = become_hafidh target_character = prev }
					}
				}
				add_to_list = characters
			}
			every_vassal = {
				limit = {
					is_physically_able_adult = yes
					exists = scope:secondary_recipient
					scope:secondary_recipient = {
						can_start_scheme = { type = become_hafidh target_character = prev }
					}
				}
				add_to_list = characters
			}
			if = {
				limit = {
					NOT = {
						exists = scope:secondary_recipient
					}
				}
				every_courtier = {
					limit = {
						is_physically_able_adult = yes
					}
					add_to_list = characters
				}
				every_vassal = {
					limit = {
						is_physically_able_adult = yes
					}
					add_to_list = characters
				}
			}
		}
	}

	populate_recipient_list = {
		scope:recipient = {
			every_courtier = {
				limit = {
					is_adult = no
					is_physically_able = yes
					age >= 6
					is_close_or_extended_family_of = scope:actor
					is_ai = yes
				}
				add_to_list = characters
			}
		}
	}

	greeting = positive
	notification_text = {
		desc = MAKE_BECOME_HAFIDH_REQUEST
	}
	
	prompt = {
		desc = SELECT_TARGET_HAFIDH_SCHEME
	}

	is_shown = {
#		has_ep1_court_positions_dlc_trigger = yes
		scope:recipient = {
			is_in_the_same_court_as = scope:actor
		}
		scope:actor = {
			AND = {
				faith.religion = religion:islam_religion
				NOT = { faith = { has_doctrine = muhammad_succession_berberist_doctrine } }
				NOT = { faith = faith:druze }
			}
		}
		trigger_if = {
			limit = {
				exists = scope:secondary_recipient
			}
			scope:secondary_recipient = {
				is_adult = no
				is_ai = yes
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			is_imprisoned = no
			NOT = { has_trait = incapable }
#			employs_court_position = court_tutor_court_position
		}
		custom_tooltip = {
			text = child_become_hafidh_not_muslim_tt
			scope:secondary_recipient.religion = religion:islam_religion
		}
		custom_tooltip = {
			text = child_become_hafidh_berberist_tt
			NOT = { scope:secondary_recipient.faith = { has_doctrine = muhammad_succession_berberist_doctrine } }
		}
		custom_tooltip = {
			text = child_become_hafidh_druze_tt
			NOT = { scope:secondary_recipient.faith = faith:druze }
		}
		trigger_if = {
			limit = {
				exists = scope:secondary_actor
			}
			scope:secondary_actor = {
				is_imprisoned = no
				NOT = { has_trait = incapable }
			}
		}
		trigger_if = {
			limit = {
				exists = scope:secondary_recipient
			}
			scope:secondary_recipient = {
				is_adult = no
				is_physically_able = yes
				age >= 6
				is_close_or_extended_family_of = scope:actor
				custom_description = {
					text = already_scheming_hafidh
					NOT = {
						any_scheme = {
							scheme_type = become_hafidh
						}
					}
				}
			}
			custom_description = {
				text = no_target_hafidh
				scope:actor = {
					OR = {
						scope:secondary_recipient = {
							can_start_scheme = { type = become_hafidh target_character = prev }
						}
						any_courtier = {
							scope:secondary_recipient = {
								can_start_scheme = { type = become_hafidh target_character = prev }
							}
						}
						any_vassal = {
							scope:secondary_recipient = {
								can_start_scheme = { type = become_hafidh target_character = prev }
							}
						}
					}
				}
			}
		}
		trigger_if = {
			limit = {
				exists = scope:secondary_actor
				exists = scope:secondary_recipient
			}
			scope:secondary_actor = {
				scope:secondary_recipient = {
					can_start_scheme = { type = become_hafidh target_character = prev }
				}
			}
		}
	}

	can_be_picked = {
		is_at_home = yes
	}
	
	# Scheme Starter Packages
	options_heading = schemes.t.agent_packages
	send_options_exclusive = yes
	## Success Chance.
	send_option = {
		flag = scheme_starter_personal_success_chance
		is_valid = {
			scope:actor.gold >= scope:actor.scheme_starter_success_chance_cost_value

		flag = scheme_starter_personal_speed

			scope:actor.gold >= scope:actor.scheme_starter_speed_cost_value

								scope:secondary_actor = court_position:court_tutor_court_position
