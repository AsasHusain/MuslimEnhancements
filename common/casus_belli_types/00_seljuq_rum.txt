	combine_into_one = yes

		NOR = {
			has_character_flag = used_lifetime_invasion

					tier = tier_county
					holder = {
						OR = {
							this = scope:defender
							target_is_liege_or_above = scope:defender
						}
					}
				}
			}
		}

		}

		piety = 25
