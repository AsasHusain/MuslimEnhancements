# Proselytize
me_proselytize_buy_conversion_gold = {
	value = root.location.county.development_level
	if = {
		limit = { var:social_type = flag:preach_nobility }
		multiply = 100
	}
	if = {
		limit = { var:social_type = flag:preach_workers }
		multiply = 50
	}
	if = {
		limit = { var:social_type = flag:preach_villagers }
		multiply = 25
	}
}
