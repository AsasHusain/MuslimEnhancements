### Legend decisions
ME_clear_ground_for_legendary_building = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_legend.dds"
	}
	decision_group_type = major
	sort_order = 100

	is_shown = {
		exists = var:ME_legendary_reward_location
		has_dlc_feature = legends
		is_landed = yes
		religion = religion:islam_religion
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = clear_ground_for_legendary_building_no_barony_tt
			any_held_title = {
				tier = tier_barony
				title_province = { has_special_building_slot = no }
				is_capital_barony = yes
				holder = root
				is_leased_out = no
			}
		}
	}

	widget = {
		gui = "decision_view_widget_option_list_generic"
		controller = decision_option_list_controller
		decision_to_second_step_button = "CHOOSE_LEGENDARY_BUILDING"
		show_from_start = yes

		item = { # Generic Legendary Mosque
			value = clear_ground_legendary_mosque
			is_valid = {
				custom_tooltip = { # The legend you finished is valid for a legendary mosque
					text = clear_ground_legendary_mosque_tt
					has_character_flag = legendary_mosque_flag
				}
			}
			current_description = clear_ground_legendary_mosque_desc
			localization = clear_ground_legendary_mosque
			icon = "gfx/interface/icons/building_types/icon_structure_cathedral_muslim.dds"
			flat = yes
			ai_chance = { value = 10 }
		}
		item = { # Ziyadid Mosque (Fictional)
			value = clear_ground_legendary_mosque_ziyadid
			is_valid = {
				custom_tooltip = { # The legend you finished is valid for a legendary mosque
					text = clear_ground_legendary_mosque_tt
					has_character_flag = legendary_mosque_flag
				}
				custom_tooltip = { # The Ziyadid mosque has not been built yet
					text = clear_ground_building_already_built
					NOT = { has_global_variable = ziyadid_mosque_built_flag }
				}
				custom_tooltip = { # Only Ziyadid dynasty
					text = clear_ground_only_ziyadid
					dynasty = dynasty:1042165 # Ziyadid
				}
			}
			current_description = clear_ground_legendary_mosque_ziyadid_desc
			localization = clear_ground_legendary_mosque_ziyadid
			icon = "gfx/interface/icons/building_types/icon_structure_imam_ali_mosque.dds"
			flat = yes
			ai_chance = { value = 10 }
		}
	}

	effect = {
		custom_tooltip = clear_ground_for_legendary_building_custom_tt
		switch = {
			trigger = yes
			scope:clear_ground_legendary_mosque = {
				set_variable = {
					name = clear_ground
					value = flag:legendary_mosque
				}
			}
			scope:clear_ground_legendary_mosque_ziyadid = {
				set_variable = {
					name = clear_ground
					value = flag:legendary_mosque_ziyadid
				}
				set_global_variable = {
					name = ziyadid_mosque_built_flag
					value = yes
				}
			}
		}
		open_view_data = {
			view = decision_detail
			data = decision:ME_pick_legendary_barony_decision
			player = root
		}
	}
	ai_check_interval = 36
	ai_potential = {
		is_at_war = no
		ai_greed < medium_positive_ai_value
		short_term_gold > ai_war_chest_desired_gold_value
		war_chest_gold >= halved_ai_war_chest_gold_maximum
	}
	ai_will_do = {
		base = 100 #AI should always reap rewards
	}
}

ME_pick_legendary_barony_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_legend.dds"
	}
	decision_group_type = major
	ai_goal = yes
	is_invisible = yes
	widget = {
		controller = create_holy_order
		barony_valid = {
			trigger_if = {
				limit = { exists = this }
				title_province = { has_special_building_slot = no }
				is_capital_barony = yes
				holder = scope:ruler
				is_leased_out = no
			}
		}
	}
	is_valid_showing_failures_only = { # Error suppressing
		has_variable = clear_ground
	}
	cost = {
	}
	effect = {
		send_interface_toast = {
			type = event_toast_effect_good
			title = clear_ground_for_legendary_building.toast
			switch = {
				trigger = root.var:clear_ground
				flag:legendary_mosque = {
					scope:barony.title_province ?= {
						add_special_building_slot = me_legendary_mosque
					}
				}
				flag:legendary_mosque_ziyadid = {
					scope:barony.title_province ?= {
						add_special_building_slot = me_legendary_mosque_ziyadid
					}
				}
			}
		}
		hidden_effect = {
			remove_variable = clear_ground
			# Shift the legend protagonist to the province
			if = {
				limit = {
					exists = var:legendary_figure
				}
				scope:barony.title_province ?= {
					set_variable = {
						name = legendary_figure
						value =  root.var:legendary_figure
					}
				}
				remove_variable = legendary_figure
			}
			# Cleanup
			if = {
				limit = {
					has_character_flag = legendary_mosque_flag
				}
				remove_character_flag = legendary_mosque_flag
			}
			if = {
				limit = {
					exists = var:ME_legendary_reward_location
				}
				remove_variable = ME_legendary_reward_location
			}
		}
	}
	
	ai_potential = {
		is_playable_character = yes
	}
	ai_will_do = {
		base = 200
	}
}
