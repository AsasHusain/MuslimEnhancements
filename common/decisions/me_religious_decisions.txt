				limit = { has_title = title:d_wahbiya }
				NOT = { exists = title:d_ibadi.holder }
			}
		}
		capital_province.county = { development_level >= 25 }
		highest_held_title_tier > 3 #Only Kings and Emperors
		piety_level >= high_piety_level

		is_capable_adult = yes
		is_imprisoned = no
		is_independent_ruler = yes

		custom_tooltip = mend_the_ibadi_schism_decision_effect_tooltip
		show_as_tooltip = {
			mend_the_ibadi_schism_scripted_effect = yes
		}
		trigger_event = {
			id = me_religious_decisions.0001
		}
	}
	
	ai_potential = {
		always = yes
	}

	ai_will_do = { #Always do it when possible.
		base = 100
		ai_value_modifier = {
			ai_zeal = 1.0
		}
		modifier = {
			factor = 0
			has_trait = cynical
		}
	}
}

#Invite a character to court who knows Arabic
invite_a_linguist_arabic = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_recruitment.dds"
	}
	ai_check_interval = 120
	sort_order = 98
	decision_group_type = courtier

	desc = invite_a_linguist_arabic_desc
	selection_tooltip = invite_a_linguist_arabic_tooltip

	is_shown = {
		# Standard filter checks.
		is_landed = yes
		has_royal_court = yes
		faith.religion = religion:islam_religion
		NOT = { faith = { has_doctrine = muhammad_succession_berberist_doctrine } }
		NOT = { faith = faith:druze }
		# Language checks.
		NOT = { # No need for linguist
			has_court_language = language_arabic 
			liege = { has_court_language = language_arabic }
			culture = { has_cultural_pillar = language_arabic  } 
		}
		any_culture_global = { # Need somewhere to recruit linguist from
			exists = culture_head
			root.liege = { has_court_language_of_culture = prev }
		}
	}

	is_valid_showing_failures_only = {
		age >= 6
		is_alive = yes
		not = { exists = involved_activity }
		is_imprisoned = no
		custom_tooltip = {
			text = scheming_already
			NOT = { # Is not currently learning a language and doesn't know court language
				any_scheme = { scheme_type = learn_language }	
			}
		}
		custom_tooltip = {
			text = linguist_already_present
			NOT = {
				any_courtier = { # You don't already have a linguist at court
					has_character_flag = court_linguist_flag
					knows_language = language_arabic
				}
			}
		}
	}

	effect = {
		save_scope_as = target_court_language
		custom_tooltip = {
			text = invite_a_linguist_tt_arabic
			trigger_event = me_religious_decisions.0007
		}
		if = {
			limit = {
				any_scheme = { is_hostile = no }
			}
			custom_description_no_bullet = { text = learn_language_will_cancel_ongoing_schemes_tt }
		}
	}

	# No cost, cost is determined in the event

	cooldown = { years = 2 } # Shouldn't have much cooldown, courtiers can die unexpectedly and the player shouldn't have to wait long to get a new one

	ai_potential = {
		is_at_war = no
		# The AI should largely try to learn its own language
		has_royal_court = yes
		# Don't do it if you already speak Arabic
		NOT = { knows_language = language_arabic }
		NOT = { has_trait = cynical }
	}
	

	ai_will_do = {
		base = 0
		modifier = {
			NOR = { # Is not currently learning a language
				knows_language = language_arabic
				any_scheme = { scheme_type = learn_language }
			}
			add = 100
		}
	}
}

# Restore the Ibadi/Wahbi/Nukkari Imamate
me_restore_ibadi_imamate_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = major

	ai_check_interval = 120

	desc = me_restore_ibadi_imamate_decision_desc
	selection_tooltip = me_restore_ibadi_imamate_decision_tooltip


	is_shown = {
		# Standard filter checks.
		exists = dynasty
		religion = religion:islam_religion
		NOT = { faith = { exists = religious_head } }
		highest_held_title_tier > 2
		OR = {
			faith.religious_head_title = title:d_ibadi
			faith.religious_head_title = title:d_wahbiya
			faith.religious_head_title = title:d_nukkari
		}
	}

	is_valid = {
		# Must be independent.
		is_independent_ruler = yes
		# Should have a decent piety level.
		piety_level >= high_piety_level
		# You CANNOT be sinful, as Ibadis believe that sinful people can't become Imams.
		custom_description = {
			text = me_restore_ibadi_imamate_decision.no_sinful_traits
			num_sinful_traits <= 0
		}
		# Now, pick any three of the valid reasons.
		calc_true_if = {
			amount >= 3

			# Having a holy site.
			custom_description = {
				text = appoint_a_righteous_caliph_decision.control_one_holy_site
				faith = {
					any_holy_site = {
						count >= 1
						holder.top_liege = root
					}
				}
			}
			# Having *2* holy sites.
			custom_description = {
				text = appoint_a_righteous_caliph_decision.control_two_holy_sites
				faith = {
					any_holy_site = {
						count >= 2
						holder.top_liege = root
					}
				}
			}
			# Having a higher-than-minimum piety level.
			piety_level >= very_high_piety_level
			# Having virtuous traits.
			custom_description = {
				text = me_restore_ibadi_imamate_decision.x_virtuous_traits
				num_virtuous_traits >= 2
			}
			# Having *even more* virtuous traits.
			custom_description = {
				text = me_restore_ibadi_imamate_decision.3_virtuous_traits
				num_virtuous_traits >= 3
			}
			# Having 5 counties of that follow your faith.
			custom_description = {
				text = me_restore_ibadi_imamate_decision.need_at_least_five_same_faith_counties
				any_sub_realm_county = {
					count >= 5
					faith = root.faith
				}
			}
			# Having 10 counties of that follow your faith.
			custom_description = {
				text = appoint_a_righteous_caliph_decision.need_at_least_ten_same_faith_counties
				any_sub_realm_county = {
					count >= appoint_a_righteous_caliph_decision_same_faith_county_count
					faith = root.faith
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		is_landed = yes
		is_available_adult = yes
		is_imprisoned = no
		is_incapable = no
		# Has to be around for the ceremony.
		not = { exists = involved_activity }
		has_contagious_deadly_disease_trigger = no
		# Has to be a valid priestly gender for the current faith.
		faith = { has_allowed_gender_for_clergy = root }
	}

	effect = {
		custom_tooltip = me_restore_ibadi_imamate_decision_effect_tooltip
		show_as_tooltip = {
			me_restore_ibadi_imamate_scripted_effect = yes
		}
		trigger_event = {
			id = me_religious_decisions.0008
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = { #Always do it when possible.
		base = 100
		ai_value_modifier = {
			ai_zeal = 1.0
		}
		modifier = {
			factor = 0
			has_trait = cynical
		}
	}
}

	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds"
	}

	selection_tooltip = me_legitimize_the_fatimid_lineage_decision_tooltip
	decision_group_type = major

	ai_check_interval = 120

	is_shown = {
		faith = {
			religious_head = root
			OR = {
				has_doctrine = muhammad_succession_shia_doctrine

				has_doctrine = islam_secondary_group_zandaqa_qarmatian_doctrine
			}
			NOT = { has_doctrine = special_doctrine_ubaydi_trait_acceptance }
		}
		OR = {
			has_trait = ubaydi
			any_close_family_member = { has_trait = ubaydi }
			any_spouse = { has_trait = ubaydi }
			any_ally = { has_trait = ubaydi }
			any_living_character = {
				has_trait = ubaydi
				OR = {
					has_relation_friend = root
					has_relation_best_friend = root
					has_relation_lover = root
					has_relation_soulmate = root
				}
			}
		}
	}

	is_valid = {
		faith = { religious_head = root }
		faith = { NOT = { has_doctrine = special_doctrine_ubaydi_trait_acceptance } }
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
	}

	effect = {
		custom_tooltip = me_legitimize_the_fatimid_lineage_decision_tooltip
		custom_tooltip = me_ubaydi_gain_opinion_tt
		hidden_effect = {
			religion:islam_religion = {
				every_faith = {
					limit = {
						religious_head = root
						OR = {
							has_doctrine = muhammad_succession_shia_doctrine
							has_doctrine = islam_secondary_group_zandaqa_ghulat_doctrine
							has_doctrine = islam_secondary_group_zandaqa_qarmatian_doctrine
						}
					}
					add_doctrine = special_doctrine_ubaydi_trait_acceptance
				}
			}
			every_living_character = {
				limit = { has_trait = ubaydi }
				add_opinion = {
					modifier = me_legitimized_fatimid_geneology_opinion
					target = root
				}
			}
		}
	}
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}

# Pick a side in the Musta'li Schism
ME_mustali_schism_convert_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds"
	}
	decision_group_type = major
	sort_order = 100

	is_shown = {
		is_landed = yes
		faith = { has_doctrine = special_doctrine_mustali_main }
	}

	is_valid_showing_failures_only = {
		is_adult = yes
		is_independent_ruler = yes
		NOT = {
			has_trait = incapable
		}
	}

	widget = {
		gui = "decision_view_widget_option_list_generic"
		controller = decision_option_list_controller
		decision_to_second_step_button = "ME_CHOOSE_MUSTALI_FAITH"
		show_from_start = yes

		# Convert to Hafizism
		item = {
			value = ME_mustali_schism_convert_hafizi_option
			is_valid = {
			}
			current_description = ME_mustali_schism_convert_hafizi_option_desc
			localization = ME_mustali_schism_convert_hafizi_option
			icon = "gfx/interface/icons/faith/hafizi.dds"

			ai_chance = {
				value = 10
				if = {
					limit = {
						NOT = { exists = title:d_hafizi.holder }
					}
					multiply = 0
				}
				if = {
					limit = {
						any_sub_realm_county = {
							faith = faith:hafizi
						}
					}
					add = 10
				}
				if = {
					limit = {
						exists = title:d_tayyibi.holder
						opinion = {
							target = title:d_tayyibi.holder
							value < -20
						}
					}
					add = 20
				}
				if = {
					limit = {
						exists = title:d_hafizi.holder
						opinion = {
							target = title:d_hafizi.holder
							value < 50
						}
					}
					add = 20
				}
			}
		}

		# Convert to Tayyibism
		item = {
			value = ME_mustali_schism_convert_tayyibi_option
			is_valid = {
			}
			current_description = ME_mustali_schism_convert_tayyibi_option_desc
			localization = ME_mustali_schism_convert_tayyibi_option
			icon = "gfx/interface/icons/faith/tayyibi.dds"

			ai_chance = {
				value = 10
				if = {
					limit = {
						NOT = { exists = title:d_tayyibi.holder }
					}
					multiply = 0
				}
				if = {
					limit = {
						any_sub_realm_county = {
							faith = faith:tayyibi
						}
					}
					add = 10
				}
				if = {
					limit = {
						exists = title:d_hafizi.holder
						opinion = {
							target = title:d_hafizi.holder
							value < -20
						}
					}
					add = 20
				}
				if = {
					limit = {
						exists = title:d_tayyibi.holder
						opinion = {
							target = title:d_tayyibi.holder
							value < 50
						}
					}
					add = 20
				}
			}
		}
		
	}

	effect = {
		if = { # Convert to Hafizism
			limit = { scope:ME_mustali_schism_convert_hafizi_option = yes }
			
			if = {
				limit = {
					capital_county.faith = faith
				}
				capital_county = { set_county_faith = faith:hafizi }
			}
			show_as_tooltip = { set_character_faith = faith:hafizi }
			hidden_effect = {
				set_character_faith_with_conversion = faith:hafizi
			}
			
			custom_tooltip = {
				text = ME_mustali_schism_convert_hafizi_percent
				every_county = {
					limit = {
						faith = { has_doctrine = special_doctrine_mustali_main }
						holder.top_liege = root
					}
					random = {
						chance = 70
						set_county_faith = faith:hafizi
					}
				}
			}
			
			faith:hafizi = {
				change_fervor = {
					value = 15
					desc = fervor_gain_mustali_convert
				}
			}
		}
		else_if = { # Convert to Tayyibism
			limit = { scope:ME_mustali_schism_convert_tayyibi_option = yes }
			
			if = {
				limit = {
					capital_county.faith = faith
				}
				capital_county = { set_county_faith = faith:tayyibi }
			}
			show_as_tooltip = { set_character_faith = faith:tayyibi }
			hidden_effect = {
				set_character_faith_with_conversion = faith:tayyibi
			}
			
			custom_tooltip = {
				text = ME_mustali_schism_convert_tayyibi_percent
				every_county = {
					limit = {
						faith = { has_doctrine = special_doctrine_mustali_main }
						holder.top_liege = root
					}
					random = {
						chance = 70
						set_county_faith = faith:tayyibi
					}
				}
			}
			
			faith:tayyibi = {
				change_fervor = {
					value = 15
					desc = fervor_gain_mustali_convert
				}
			}
		}
		
	}
	
	cost = {
		piety = {
			value = major_piety_value
		}
	}
	
	ai_check_interval = 12
	ai_potential = {
		highest_held_title_tier >= tier_county
		is_landed = yes
		is_at_war = no
	}
	ai_will_do = {
		base = 100
	}
}
